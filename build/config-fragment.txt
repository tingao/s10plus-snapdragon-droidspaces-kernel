# --- netfilter match required by dockerd/iptables inside the Droidspaces container ---
# Stock Samsung kernel shipped this =y; it was missing from our baseline build and
# found live on-device: dockerd fails with
#   iptables -t nat -A PREROUTING -m addrtype --dst-type LOCAL -j DOCKER
#   "Extension addrtype revision 0 not supported / No chain/target/match by that name"
# which then breaks Portainer and DNAT.
CONFIG_NETFILTER_XT_MATCH_ADDRTYPE=y

# --- IPv6 NAT/masquerade parity with the stock Samsung kernel (droidspaces has IPv6 NAT) ---
# Only the two masquerade symbols were missing; IP6_NF_IPTABLES/IP6_NF_NAT/NF_NAT_IPV6 were
# already =y. Deliberately NOT enabling CONFIG_NF_TABLES: leaving nf_tables out removes the
# whole nf_tables CVE class from this kernel (intentional security divergence from stock).
CONFIG_NF_NAT_MASQUERADE_IPV6=y
CONFIG_IP6_NF_TARGET_MASQUERADE=y
